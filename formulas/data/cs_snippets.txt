@@ Searching and sorting|Binary search|python|p|Find a value in a sorted list by halving the range.
def search(a, x):
    lo, hi = 0, len(a) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if a[mid] == x:
            return mid
        if a[mid] < x:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1
print(search([1, 3, 5, 7, 9], 7))
##
3
@@ Searching and sorting|Binary search|java|p|Find a value in a sorted array by halving the range.
public class Main {
    static int search(int[] a, int x) {
        int lo = 0, hi = a.length - 1;
        while (lo <= hi) {
            int mid = (lo + hi) / 2;
            if (a[mid] == x) return mid;
            if (a[mid] < x) lo = mid + 1; else hi = mid - 1;
        }
        return -1;
    }
    public static void main(String[] args) {
        System.out.println(search(new int[]{1, 3, 5, 7, 9}, 7));
    }
}
##
3
@@ Searching and sorting|Bubble sort|python|p|Repeatedly swap neighbours that are out of order.
a = [5, 2, 9, 1]
for i in range(len(a)):
    for j in range(len(a) - i - 1):
        if a[j] > a[j + 1]:
            a[j], a[j + 1] = a[j + 1], a[j]
